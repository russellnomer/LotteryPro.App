Since you’re in New York, you have perfect free official data to power it—no scraping needed for the core prize info, and no reliance on unreliable “patterns” in serial numbers (as we discussed before).
The app can help users choose games by showing:
	•	Which games still have big prizes left
	•	Total remaining prize money in each game
	•	Simple “value scores” (e.g., remaining prize pool relative to ticket price)
	•	Filters by price, top prizes left, etc.
	•	Sortable rankings so you instantly see the “best” current games
Best Data Source (Free & Official)
New York publishes a daily-updated open dataset:
	•	Endpoint: https://data.ny.gov/resource/nzqa-7unk.json
	•	It’s a JSON array with one row per prize tier per game.
	•	Fields include: game_number, game_name, prize_amount (e.g., “$500,000” or “$10,000/YR/20YRS”), unpaid (remaining prizes), paid, total.
This lets you calculate real-time metrics like:
	•	Remaining winning tickets per game (sum(unpaid))
	•	Total remaining prize value (sum unpaid × prize value)
	•	% of prizes still available (rough proxy for how “fresh” the game is)
Ticket prices and top-prize details aren’t in this dataset, but you can hardcode a small lookup dict (games don’t change price often) or add light scraping later if you want.

import streamlit as st
import requests
import pandas as pd
from datetime import datetime

st.title("🗽 NY Scratch-Off Helper")
st.markdown("**Real-time remaining prizes from official NY data** — Pick smarter!")

# Fetch latest data
@st.cache_data(ttl=3600)  # Refresh hourly
def load_data():
    url = "https://data.ny.gov/resource/nzqa-7unk.json"
    response = requests.get(url)
    df = pd.DataFrame(response.json())
    return df

df = load_data()

# Simple prize value parser (handles $, commas, and rough annuity approx)
def parse_prize_amount(amount_str):
    if not amount_str or amount_str == "Free Ticket":
        return 0.0
    # Remove $ and commas
    s = amount_str.replace("$", "").replace(",", "").strip()
    try:
        if any(x in s.upper() for x in ["YR", "WK", "LIFE", "DAY"]):
            # Rough annuity value (common approximations)
            if "1,000/WK" in amount_str or "WK/LIFE" in amount_str:
                return 1_000_000  # ~$1M lifetime value
            elif "YR/20" in amount_str:
                num = float(''.join(filter(str.isdigit, s.split('/')[0])))
                return num * 20 * 0.8  # conservative
            return float(''.join(filter(str.isdigit, s))) * 10  # fallback
        else:
            return float(s)
    except:
        return 0.0

# Process data
df['prize_value'] = df['prize_amount'].apply(parse_prize_amount)
df['unpaid'] = pd.to_numeric(df['unpaid'], errors='coerce').fillna(0)

# Group by game
grouped = df.groupby(['game_number', 'game_name']).agg(
    total_remaining_winners=('unpaid', 'sum'),
    total_remaining_value=('unpaid', lambda x: (x * df.loc[x.index, 'prize_value']).sum()),
    top_prize_remaining=('unpaid', 'max'),  # rough - we'll refine with actual top
    prize_tiers=('prize_amount', 'nunique')
).reset_index()

# Add your price lookup here (expand from official site)
price_lookup = {
    "1464": 30,   # $10,000,000 BONUS example
    "1659": 1,    # TRIPLE MATCH
    "1661": 5,    # CASINO NIGHTS
    # Add more: game_number: price
    # Example: "1657": 20,
}
grouped['price'] = grouped['game_number'].map(price_lookup).fillna(5)  # default fallback

# Value metrics
grouped['value_per_dollar'] = grouped['total_remaining_value'] / (grouped['price'] * grouped['total_remaining_winners'].replace(0, 1))
grouped['big_prizes_left'] = df[df['prize_value'] > 10000].groupby(['game_number', 'game_name'])['unpaid'].sum().reindex(grouped.set_index(['game_number', 'game_name']).index, fill_value=0).values

# Filters & display
col1, col2 = st.columns(2)
with col1:
    min_price = st.slider("Min ticket price", 1, 30, 1)
with col2:
    min_big = st.slider("Min big prizes left ($10k+)", 0, 50, 1)

filtered = grouped[
    (grouped['price'] >= min_price) &
    (grouped['big_prizes_left'] >= min_big)
].sort_values('total_remaining_value', ascending=False)

st.dataframe(
    filtered[['game_name', 'price', 'big_prizes_left', 'total_remaining_value', 'total_remaining_winners', 'value_per_dollar']],
    column_config={
        "total_remaining_value": st.column_config.NumberColumn("Remaining Prize $", format="$%.0f"),
        "value_per_dollar": st.column_config.NumberColumn("Value Score", format="%.3f"),
    },
    use_container_width=True,
    hide_index=True
)

st.caption(f"Data refreshed: {datetime.now().strftime('%Y-%m-%d %H:%M')} | Official NY source • For entertainment only")
